import numpy as np
from problems import BudgetExhausted

def optimize(f, dim, bounds, budget, seed):
    np.random.seed(seed)
    low, high = bounds
    inertia = 0.9
    inertia_decay = 0.99
    cognitive = 1.5
    social = 1.5
    particles = 30
    max_iterations = 1000
    stagnation = 50

    # Initialize particles
    positions = np.random.uniform(low, high, (particles, dim))
    velocities = np.random.uniform(-1, 1, (particles, dim))
    pbest_positions = positions.copy()
    pbest_scores = np.array([f(x) for x in positions])
    gbest_position = pbest_positions[np.argmin(pbest_scores)]
    gbest_score = np.min(pbest_scores)

    evaluations = 0
    best_score = np.inf
    best_position = None
    stagnation_count = 0

    for _ in range(max_iterations):
        if evaluations >= budget:
            raise BudgetExhausted()

        for i in range(particles):
            if evaluations >= budget:
                raise BudgetExhausted()

            # Update velocity
            r1 = np.random.rand(dim)
            r2 = np.random.rand(dim)
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (pbest_positions[i] - positions[i]) +
                            social * r2 * (gbest_position - positions[i]))

            # Update position
            positions[i] = np.clip(positions[i] + velocities[i], low, high)

            # Evaluate new position
            current_score = f(positions[i])
            evaluations += 1

            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = positions[i]

        # Update global best
        current_gbest_score = np.min(pbest_scores)
        if current_gbest_score < gbest_score:
            gbest_score = current_gbest_score
            gbest_position = pbest_positions[np.argmin(pbest_scores)]
            best_score = gbest_score
            best_position = gbest_position
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Decay inertia
        inertia *= inertia_decay

        # Check stagnation
        if stagnation_count >= stagnation:
            # Restart by reinitializing particles
            positions = np.random.uniform(low, high, (particles, dim))
            velocities = np.random.uniform(-1, 1, (particles, dim))
            pbest_positions = positions.copy()
            pbest_scores = np.array([f(x) for x in positions])
            gbest_position = pbest_positions[np.argmin(pbest_scores)]
            gbest_score = np.min(pbest_scores)
            stagnation_count = 0

    return {
        'position': best_position,
        'score': best_score,
        'evaluations': evaluations
    }
